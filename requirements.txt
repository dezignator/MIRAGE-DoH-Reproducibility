# =============================================================================
# MIRAGE-DoH Reproducibility Artifact — Python dependencies
# =============================================================================
#
# Pinned for reproducibility. Every version below is the exact version used to
# produce the numbers in Chapter 3 of the thesis (Tables 5-12, Figures 1-6).
# All packages are installed from PyPI under their original licences.
#
# Python version required: >= 3.9, < 3.12
#   - 3.9, 3.10, 3.11 are all tested.
#   - 3.12 is not tested because pycryptodome has not stabilised on it.
#
# Install with:
#     pip install -r requirements.txt
#     pip install -e .          # editable install of src/ (see setup.py)
#
# For a fully isolated conda environment (recommended for the full 32 GB
# reproduction), use environment.yml instead of this file.
#
# Google Colab: torch, numpy, pandas, scipy, scikit-learn and matplotlib are
# pre-installed at compatible versions. Running "pip install -r
# requirements.txt" in Colab will print "Requirement already satisfied" for
# those; this is expected. The packages that pip will actually install in
# Colab are: gudhi, ripser, persim, snntorch, pycryptodome, dask, and
# python-igraph.
# =============================================================================


# -----------------------------------------------------------------------------
# 1. Core scientific computing
#    Used everywhere. Numpy/Scipy for array math; pandas for dataset tables.
#    numpy is pinned to 1.x because 2.0 changed the ABI and several of our
#    dependencies (ripser, gudhi) had not yet released 2.x-compatible wheels
#    at the time the thesis numbers were produced.
# -----------------------------------------------------------------------------
numpy==1.26.4
scipy==1.13.1
pandas==2.2.2


# -----------------------------------------------------------------------------
# 2. Deep learning
#    torch powers the CMA neuroevolution (Eq. 1-5), the TMN LSTM-attention
#    (Eq. 9), the SGI Deep Q-Network (Eq. 19), and the CNN-LSTM baseline.
#    torchvision is only used by the CNN-LSTM baseline (Chapter 3, Table 8).
#    If you have a CUDA-capable GPU and want the CUDA wheel, replace the line
#    below with the version-matched wheel from https://pytorch.org/get-started
# -----------------------------------------------------------------------------
torch==2.3.1
torchvision==0.18.1
torchaudio==2.3.1


# -----------------------------------------------------------------------------
# 3. Classical machine learning
#    scikit-learn provides the Random Forest baseline (Table 8), the
#    calibration utilities, and the cross-validation splitters used in the
#    hyperparameter sweeps of Table 5 and Figure 5.
#    joblib is required by scikit-learn's parallel backend and by our
#    multi-process CMA dispatcher (Eq. 31).
# -----------------------------------------------------------------------------
scikit-learn==1.5.0
joblib==1.4.2


# -----------------------------------------------------------------------------
# 4. Topological data analysis (TMN — Eq. 6-10 and Figure 4)
#    gudhi:       Simplicial complex construction (Vietoris-Rips, Čech).
#    ripser:      Fast persistent homology (Betti-0, Betti-1, higher).
#    persim:      Persistence diagram distance metrics (Wasserstein,
#                 bottleneck) used in Table 9 and Figure 2.
#    All three are pure-Python or ship manylinux wheels; no compilation
#    needed on Linux/macOS. On Windows, gudhi may need Microsoft Build Tools.
# -----------------------------------------------------------------------------
gudhi==3.10.0
ripser==0.6.12
persim==0.3.7


# -----------------------------------------------------------------------------
# 5. Neuromorphic simulation (hardware_simulation/loihi_mock.py)
#    snntorch is the CPU-friendly spiking neural network library used to
#    reproduce the LIF neuron dynamics (Eq. 44-45) and the spike-encoding
#    scheme (Eq. 43) reported in Chapter 3 for Intel Loihi 2.
#    brian2 is an alternative simulator used for cross-checking; it is
#    optional and can be removed if you only need snntorch.
#    If you have access to real Intel Loihi 2 hardware, install "nxsdk"
#    from Intel instead (not on PyPI; requires INRC membership).
# -----------------------------------------------------------------------------
snntorch==0.9.1
brian2==2.7.1


# -----------------------------------------------------------------------------
# 6. Graph analysis (SGI — Eq. 17-21 and Figure 6)
#    networkx provides the graph Laplacian, spectral embedding and the DQN
#    state-space construction. python-igraph is used in the alternative
#    implementation path that produced the 63.7 % / 99.2 % spectral precision
#    comparison in Table 9 and Table 12.
# -----------------------------------------------------------------------------
networkx==3.3
python-igraph==0.11.5


# -----------------------------------------------------------------------------
# 7. Cryptographic references (MCS — Eq. 11-13)
#    pycryptodome provides the reference implementations of X25519 and AES-GCM
#    used in the shard-combination tests of Table 5. The post-quantum KEMs
#    (ML-KEM-768, ML-KEM-1024, Classic McEliece) are not in pycryptodome; they
#    are referenced through published test vectors fetched by
#    data/download_datasets.py --pqc-vectors and verified by
#    tests/test_math_utils.py::test_kem_combiner. This is a deliberate design:
#    the artifact does not reimplement post-quantum KEMs.
# -----------------------------------------------------------------------------
pycryptodome==3.20.0


# -----------------------------------------------------------------------------
# 8. Visualisation (Figures 1-6)
#    matplotlib and seaborn produce the six figures. Graphviz (via the
#    "graphviz" Python package) produces the architecture diagram of Figure 1
#    from figures_src/figure_1.dot.
#    The system-level graphviz binary must also be installed:
#       Ubuntu/Debian: sudo apt-get install graphviz
#       macOS:         brew install graphviz
#       Colab:         sudo apt-get install -y graphviz
# -----------------------------------------------------------------------------
matplotlib==3.9.0
seaborn==0.13.2
graphviz==0.20.3


# -----------------------------------------------------------------------------
# 9. Configuration
#    PyYAML parses the eight configs/*.yaml files.
# -----------------------------------------------------------------------------
PyYAML==6.0.1


# -----------------------------------------------------------------------------
# 10. Data handling and parallel preprocessing
#     requests is used by data/download_datasets.py with HTTP resume support.
#     dask is the parallel preprocessing framework mentioned in Chapter 6 of
#     the thesis (Section 6.13); it is also used here for the Dask-based
#     preprocessing of the 32 GB corpus in data/preprocess_*.py so that the
#     full reproduction pipeline matches the thesis pipeline exactly.
# -----------------------------------------------------------------------------
requests==2.32.3
dask==2024.6.0
dask[dataframe]==2024.6.0


# -----------------------------------------------------------------------------
# 11. Testing
#     pytest runs tests/test_math_utils.py, test_cli_tools.py and
#     test_reproduce_tables.py. pytest-cov produces the coverage report
#     referenced in REPRODUCIBILITY_CHECKLIST.md §11.
# -----------------------------------------------------------------------------
pytest==8.2.2
pytest-cov==5.0.0


# -----------------------------------------------------------------------------
# 12. Utilities
#     tqdm: progress bars in download and training scripts.
#     packaging: version parsing for the compatibility check in
#     src/utils/seed.py (refuses to run on Python < 3.9).
#     psutil: memory measurement in the toy-vs-full comparison of
#     REPRODUCIBILITY_CHECKLIST.md §3.
# -----------------------------------------------------------------------------
tqdm==4.66.4
packaging==24.1
psutil==6.0.0
